local function HttpGet(url)
    if syn and type(syn.request) == "function" then
        local res = syn.request({ Url = url, Method = "GET" })
        if res and res.Body then return res.Body end
    end
    if type(request) == "function" then
        local res = request({ Url = url, Method = "GET" })
        if res and res.Body then return res.Body end
    end
    if type(game.HttpGet) == "function" then
        return game:HttpGet(url)
    end
    error("No HTTP method available in this executor")
end

-- HELPERS
local function Create(class, properties, parent)
    local object = Instance.new(class)

    for property, value in pairs(properties or {}) do
        object[property] = value
    end

    object.Parent = parent
    return object
end

local function Corner(parent, radius)
    return Create("UICorner", {
        CornerRadius = UDim.new(0, radius or 8)
    }, parent)
end

local function Stroke(parent, color, thickness, transparency)
    return Create("UIStroke", {
        Color = color or COLORS.Border,
        Thickness = thickness or 1,
        Transparency = transparency or 0
    }, parent)
end

local function Label(parent, text, size, position, font, textSize, color)
    return Create("TextLabel", {
        BackgroundTransparency = 1,
        Text = text,
        Size = size,
        Position = position,
        Font = font or Enum.Font.Gotham,
        TextSize = textSize or 12,
        TextColor3 = color or COLORS.White,
        TextXAlignment = Enum.TextXAlignment.Left,
        TextYAlignment = Enum.TextYAlignment.Center
    }, parent)
end

local function Tween(object, duration, properties)
    local tween = TweenService:Create(
        object,
        TweenInfo.new(duration, Enum.EasingStyle.Quart, Enum.EasingDirection.Out),
        properties
    )

    tween:Play()
    return tween
end

local function UpdateScale()
    local camera = workspace.CurrentCamera
    if not camera then return end

        math.min(viewport.X / 420, viewport.Y / 390),
        0.72,
        1
    )

    Main:FindFirstChildOfClass("UIScale").Scale = scale

-- DRAGGING
local dragging = false
local dragInput
local dragStart
local startPosition

local function UpdateDrag(input)
    local delta = input.Position - dragStart

    Main.Position = UDim2.new(
        startPosition.X.Scale,
        startPosition.X.Offset + delta.X,
        startPosition.Y.Scale,
        startPosition.Y.Offset + delta.Y
    )
end

local DragArea = Create("Frame", {
    Size = UDim2.new(1, -90, 0, 65),
    BackgroundTransparency = 1,
    Active = true
}, Main)

DragArea.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then

        dragging = true
        dragStart = input.Position
        startPosition = Main.Position

        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                dragging = false
            end
        end)
    end
end)

DragArea.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch then
        dragInput = input
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if dragging and input == dragInput then
        UpdateDrag(input)
    end
end)

-- SELECTION STATE
local Selected = false
local Buttons = {}

local function SetButtonState(name)
    for buttonName, data in pairs(Buttons) do
        local active = buttonName == name

        Tween(data.Button, 0.15, {
            BackgroundColor3 = active and COLORS.Hover or COLORS.Secondary
        })

        data.Border.Color = active and COLORS.White or COLORS.Border
        data.Border.Thickness = active and 1.5 or 1
        data.Bar.BackgroundTransparency = active and 0 or 1
    end
end

-- GUI EXIT
local function CloseGUI()
    if ScreenGui and ScreenGui.Parent then
        ScreenGui:Destroy()
    end
end

-- SELECTION CALLBACKS
local function OnSelect(version)
    if Selected then return end
    Selected = true

    CONFIG.Version = version
    SetButtonState(version)
    SetStatus("Fetching " .. version .. "...", COLORS.White)

    -- hard input lock + instant disappear so no double clicks can queue a second load
    for _, data in pairs(Buttons) do
        data.Button.Active = false
        data.Button.Interactable = false
        data.Button.AutoButtonColor = false
    end
    if ScreenGui and ScreenGui.Parent then
        ScreenGui.Enabled = false
    end

    local url = (version == "Lite") and LITE_URL or NORMAL_URL

    task.spawn(function()
        local ok, err = pcall(function()
            local source = HttpGet(url)
            if type(source) ~= "string" or source == "" then
                error("empty response from server")
            end
            local fn, loadErr = loadstring(source)
            if not fn then error(loadErr or "failed to compile") end
            fn()
        end)

        if ok then
            CloseGUI()
        else
            -- put the picker back so the error is visible and a retry is possible
            if ScreenGui and ScreenGui.Parent then
                ScreenGui.Enabled = true
            end
            for _, data in pairs(Buttons) do
                data.Button.Active = true
                data.Button.Interactable = true
            end
            Selected = false
            SetStatus("Error: " .. tostring(err), COLORS.Red)
        end
    end)
end

        OnSelect(name)
    end)

        Button = Button,
        Border = Border,

    }
