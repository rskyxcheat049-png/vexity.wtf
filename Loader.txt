	Flow: key system -> if the game is supported, a single vexity notification
	appears and the game script loads instantly (no loader interface at all).
	If the game is not supported, a small vexity-themed dialog asks whether to

	alternate places like pro servers / ranked / updated root places still count
	as supported.

local notifyToken = 0
local notifyBarTween

local function VexityNotify(text, kind)
	notifyToken += 1
	local token = notifyToken

	local color = Color3.fromHex("#FFFFFF")
	local icon = "i"
	if kind == "good" then
		color = Color3.fromHex("#6FCF97")
		icon = "✓"
	elseif kind == "bad" then
		icon = "!"
	elseif kind == "warning" then
		color = Color3.fromHex("#E0A458")
		icon = "!"
	end

	notifyIcon.Text = icon
	notifyIcon.TextColor3 = color
	notifyText.Text = tostring(text or "")
	notifyBar.BackgroundColor3 = color
	notifyBar.BackgroundTransparency = 0.15
	notifyBar.Size = UDim2.new(1, 0, 0, 2)

	if notifyBarTween then
		notifyBarTween:Cancel()
	end
	notifyBarTween = NotifyTweenService:Create(
		notifyBar,
		TweenInfo.new(2.5, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
		{ Size = UDim2.new(0, 0, 0, 2) }
	)
	notifyBarTween:Play()

	NotifyTweenService:Create(
		notifyToast,
		TweenInfo.new(0.35, Enum.EasingStyle.Back, Enum.EasingDirection.Out),
		{ Position = UDim2.new(0.5, 0, 0.955, 0) }
	):Play()

	task.delay(2.7, function()
		if token ~= notifyToken then return end
		NotifyTweenService:Create(
			notifyToast,
			TweenInfo.new(0.28, Enum.EasingStyle.Quad, Enum.EasingDirection.In),
			{ Position = UDim2.new(0.5, 0, 1.055, 0) }
		):Play()
	end)
end

local placeId = game.PlaceId
local universeId = tonumber(game.GameId) or 0
local gameName = tostring((game.Name ~= "" and game.Name) or "Unknown Game")

-- GetProductInfo is a YIELDING call. In some games (Rivals especially, once
-- the loader is obfuscated) it can hang forever and kill the whole startup,
-- which is why nothing - not even the intro/notifications - appeared.
-- It is only used to prettify the displayed name, so run it detached and
-- never let it block the intro.
task.spawn(function()
	local ok, info = pcall(function()
		return Marketplace:GetProductInfo(placeId)
	end)
	if ok and info and type(info.Name) == "string" and info.Name ~= "" then
		gameName = info.Name
	end
end)

local function normalizeName(value)
	return tostring(value or ""):lower():gsub("[^%w%s]", " "):gsub("%s+", " ")
end

local matched
for _, g in ipairs(GameList) do
	if g.PlaceId == placeId or (universeId > 0 and g.UniverseId == universeId) then
		matched = g
		break
	end
end
if not matched then
	local current = normalizeName(gameName)
	for _, g in ipairs(GameList) do
		for _, alias in ipairs(g.Names) do
			if string.find(current, alias, 1, true) then
				matched = g
				break
			end
		end
		if matched then break end
	end
end

local function loadScript(url, label)
	if type(url) ~= "string" or url == "" then return end
	task.spawn(function()
		local ok, fn = pcall(function() return loadstring(game:HttpGet(url)) end)
		if ok and type(fn) == "function" then
			task.spawn(function() pcall(fn) end)
		else
			VexityNotify("Failed to load " .. label, "bad")
		end
	end)
end

local function secureParent()
	local ok, hidden = pcall(function()
		if gethui then return gethui() end
		if get_hidden_gui then return get_hidden_gui() end
	end)
	if ok and hidden then return hidden end
	return game:GetService("CoreGui")
end

local loaderDone = false
local function finishLoader()
	if loaderDone then return end
	loaderDone = true
	if matched then
		VexityNotify(gameName .. " is supported — loading script", "good")
		loadScript(matched.Script, matched.Name)
	else
		showUniversalPrompt()
	end
end

print("[vexity] loader start | place=" .. tostring(placeId)
	.. " universe=" .. tostring(universeId)
	.. " name=" .. gameName
	.. " matched=" .. tostring(matched and matched.Name or "none"))

-- Never fail silently: if the intro errors (obfuscator can mangle tweens or
-- task.spawn callbacks), report it and continue straight to loading.
task.spawn(function()
	local ok, err = pcall(playIntro, finishLoader)
	if not ok then
		warn("[vexity] intro failed: " .. tostring(err))
		finishLoader()
	end
end)

-- Watchdog: if for any reason the intro never reaches its callback, load the
-- script anyway after 8s so Rivals (or any game) is never left with nothing.
task.delay(8, function()
	if loaderDone then return end
	warn("[vexity] loader watchdog fired — forcing script load")
	finishLoader()
end)

			local old = secureParent():FindFirstChild("VexityUnsupported")
			if old then old:Destroy() end
		end)
	end,
}
end

local function cleanName(raw)
	if type(raw) ~= "string" then
		return ""
	end
	local value = raw
	value = value:gsub("%b[]%s*", "")
	value = value:gsub("[^\32-\126]", "")
	value = value:gsub("%s+", " ")
	value = value:gsub("^%s+", ""):gsub("%s+$", "")
	return value
end

local function formatCount(value)
	local n = tonumber(value)
	if not n or n < 0 then
		return "--"
	end
	if n >= 1e9 then
		return (string.format("%.1f", n / 1e9):gsub("%.0$", "")) .. "B"
	elseif n >= 1e6 then
		return (string.format("%.1f", n / 1e6):gsub("%.0$", "")) .. "M"
	elseif n >= 1e3 then
		return (string.format("%.1f", n / 1e3):gsub("%.0$", "")) .. "K"
	end
	return tostring(math.floor(n))
end

local function initials(name)
	local words = {}
	for word in tostring(name or ""):gmatch("%a+") do
		words[#words + 1] = word
	end
	if #words >= 2 then
		return (words[1]:sub(1, 1) .. words[2]:sub(1, 1)):upper()
	elseif #words == 1 then
		return words[1]:sub(1, 2):upper()
	end
	return "?"
end

end

local snowAccumulator = 0
local snowConnection
snowConnection = RunService.Heartbeat:Connect(function(delta)
	if not ScreenGui.Parent then
		if snowConnection then
			snowConnection:Disconnect()
		end
		return
	end

	snowAccumulator += math.min(delta, 0.1)
	if snowAccumulator < 1 / 30 then
		return
	end
	local step = snowAccumulator
	snowAccumulator = 0

	for _, flake in ipairs(flakes) do
		flake.phase += step
		local x = flake.frame.Position.X.Scale + (flake.drift + math.sin(flake.phase * 2) * 0.012) * step * 4
		local y = flake.frame.Position.Y.Scale + flake.speed * step
		if y > 1.05 then
			y = rng:NextNumber(-0.15, -0.05)
			x = rng:NextNumber(0, 1)
		end
		flake.frame.Position = UDim2.new(x, 0, y, 0)
	end
end)

local function httpJSON(url)
	local ok, body = pcall(function()
		return game:HttpGet(url, true)
	end)
	if not ok or type(body) ~= "string" then
		ok, body = pcall(function()
			return game:HttpGet(url)
		end)
	end
	if not ok then
		return nil
	end
	if type(body) == "table" then
		return body
	end
	if type(body) ~= "string" or body == "" then
		return nil
	end
	local decoded, data = pcall(function()
		return HttpService:JSONDecode(body)
	end)
	if decoded then
		return data
	end
	return nil
end

local function fetchGameData()
	for _, gameInfo in ipairs(GAMES) do
		if not gameInfo.universeId and gameInfo.placeId then
			local resolved = httpJSON(
				"https://apis.roblox.com/universes/v1/places/"
					.. tostring(gameInfo.placeId)
					.. "/universe"
			)
			if resolved and tonumber(resolved.universeId) then
				gameInfo.universeId = tonumber(resolved.universeId)
			end
		end
		if gameInfo.universeId then
			local view = gameViews[gameInfo.placeId]
			if view then
				setIcon(view.icon, iconURL(gameInfo.universeId))
			end
		end
	end

	local universeIds = {}
	for _, gameInfo in ipairs(GAMES) do
		if gameInfo.universeId then
			universeIds[#universeIds + 1] = tostring(gameInfo.universeId)
		end
	end
	if #universeIds == 0 then
		return
	end

	local payload = httpJSON(
		"https://games.roblox.com/v1/games?universeIds=" .. table.concat(universeIds, ",")
	)
	if not payload or type(payload.data) ~= "table" then
		return
	end

	dataLoaded = true
	local byId = {}
	for _, info in ipairs(payload.data) do
		byId[tonumber(info.id)] = info
	end

	for _, gameInfo in ipairs(GAMES) do
		local info = gameInfo.universeId and byId[tonumber(gameInfo.universeId)]
		if info then
			local cleaned = cleanName(info.name)
			if cleaned ~= "" then
				gameInfo.liveName = cleaned
			end
			if info.creator and type(info.creator.name) == "string" then
				gameInfo.creator = info.creator.name
			end
			gameInfo.playing = tonumber(info.playing) or gameInfo.playing
			updateGameRow(gameInfo)
		end
	end
end

task.spawn(function()
	local ok, err = pcall(fetchGameData)
	if not ok then
		warn("[vexity.wtf] game data: " .. tostring(err))
	end
	if ScreenGui.Parent then
		updateCard()
		updateGamesFooter()
	end
end)

local busy = false

local function normalizeKey(value)
	return tostring(value or ""):gsub("%s+", ""):lower()
end

local function isKeyValid(value)
	local attempt = normalizeKey(value)
	for key, valid in pairs(CONFIG.ValidKeys) do
		if valid and normalizeKey(key) == attempt then
			return true
		end
	end
	return false
end

local function loadSavedKey()
	if type(isfile) ~= "function" or type(readfile) ~= "function" then
		return nil
	end
	local ok, exists = pcall(isfile, CONFIG.SaveFile)
	if not ok or not exists then
		return nil
	end
	local readOk, contents = pcall(readfile, CONFIG.SaveFile)
	if not readOk or type(contents) ~= "string" then
		return nil
	end
	return contents:gsub("^%s+", ""):gsub("%s+$", "")
end

local function saveKey(value)
	if type(writefile) ~= "function" then
		return
	end
	pcall(writefile, CONFIG.SaveFile, tostring(value))
end

local function setBusy(state)
	busy = state
	SubmitBtn.Text = state and "CHECKING" or "ENTER"
	tween(SubmitBtn, 0.15, {
		BackgroundColor3 = state and T.Surface3 or T.White,
	})
	SubmitBtn.TextColor3 = state and T.Muted or T.Bg
	Shine.Visible = not state
	if state then
		-- Fail-safe: never leave the key stuck on "CHECKING" if a thread dies mid-check.
		busyToken += 1
		local token = busyToken
		task.delay(9, function()
			if not (busy and busyToken == token) then
				return
			end
			if ScreenGui.Parent then
				pcall(setBusy, false)
			else
				busy = false
			end
		end)
	end
end

local function shakePanel()
	local base = Panel.Position
	for index = 1, 6 do
		local direction = (index % 2 == 0) and 1 or -1
		Panel.Position = base + UDim2.fromOffset(direction * (10 - index), 0)
		task.wait(0.045)
	end
	Panel.Position = base
end

local function tryLoadMainScript()
	local url = CONFIG.MainScriptURL
	if type(url) ~= "string" or url == "" then
		local env = getgenv and getgenv() or nil
		if env and type(env.VexityMainScriptURL) == "string" then
			url = env.VexityMainScriptURL
		end
	end

	if type(url) ~= "string" or url == "" then
		print("[vexity.wtf] Key accepted. Main script URL is empty.")
		return
	end

	local ok, err = pcall(function()
		local source = game:HttpGet(url)
		local fn, compileError = loadstring(source)
		if not fn then
			error(compileError)
		end
		fn()
	end)
	if not ok then
		warn("[vexity.wtf] main script failed: " .. tostring(err))
	end
end

local function attemptKey(rawValue)
	if busy or not ScreenGui.Parent then
		return
	end

	local entered = tostring(rawValue or "")
	entered = entered:gsub("^%s+", ""):gsub("%s+$", "")

	if normalizeKey(entered) == "" then
		notify("Enter your key first", "info")
		if KeyBox.Parent then
			KeyBox:CaptureFocus()
		end
		return
	end

	setBusy(true)
	tween(Panel, 0.18, { Size = UDim2.fromOffset(452, 372) },
		Enum.EasingStyle.Back, Enum.EasingDirection.Out)

	local dots = { "CHECKING", "CHECKING .", "CHECKING . .", "CHECKING . . ." }
	for index = 1, 9 do
		if not ScreenGui.Parent then
			return
		end
		SubmitBtn.Text = dots[((index - 1) % 4) + 1]
		task.wait(0.13)
	end

	tween(Panel, 0.22, { Size = UDim2.fromOffset(BASE_WIDTH, BASE_HEIGHT) },
		Enum.EasingStyle.Back, Enum.EasingDirection.Out)
	task.wait(0.2)

	if isKeyValid(entered) then
		saveKey(entered)
		SubmitBtn.Text = "Authorized"
		SubmitBtn.BackgroundColor3 = T.White
		SubmitBtn.TextColor3 = T.Bg
		Shine.Visible = false
		notify("Key accepted — welcome to vexity.wtf", "good")
		task.wait(0.85)

		if snowConnection then
			snowConnection:Disconnect()
		end
		tween(Panel, 0.35, {
			Position = Panel.Position + UDim2.fromOffset(0, 14),
			BackgroundTransparency = 1,
		}, Enum.EasingStyle.Quad, Enum.EasingDirection.In)
		tween(PanelStroke, 0.35, { Transparency = 1 })
		tween(Backdrop, 0.35, { BackgroundTransparency = 1 })
		task.wait(0.32)

		for _, blur in ipairs(blurs) do
			tween(blur, 0.3, { Size = 0 }, Enum.EasingStyle.Quad, Enum.EasingDirection.In)
		end
		task.delay(0.4, function()
			for _, blur in ipairs(blurs) do
				if blur.Parent then
					blur:Destroy()
				end
			end
		end)

		ScreenGui:Destroy()
		runVexityLoader()
		return
	end

	print("[vexity.wtf] key check: typed=" .. normalizeKey(entered)
		.. " | accepted=vexitywowcursedxd | result=REJECTED")
	notify("Invalid key — try again", "bad")
	tween(KeyStroke, 0.12, { Color = T.StrokeHi })
	task.delay(0.55, function()
		if KeyStroke.Parent then
			tween(KeyStroke, 0.3, { Color = T.Stroke })
		end
	end)
	shakePanel()
	setBusy(false)
end

_G.__vexityAttempt = attemptKey

-- Run attempts on a scheduler thread so every task.wait inside attemptKey can
-- always resume, even when triggered from unusual input contexts.
local function requestAttempt(value)
	task.spawn(attemptKey, value)
end

SubmitBtn.MouseButton1Click:Connect(function()
	requestAttempt(KeyBox.Text)
end)
KeyBox.FocusLost:Connect(function(enterPressed)
	if enterPressed then
		requestAttempt(KeyBox.Text)
	end
end)
